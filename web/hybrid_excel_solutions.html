Solution 1: ExcelJS + Canvas 하이브리드

// ExcelJS + Canvas 하이브리드 렌더링 엔진 class ExcelCanvasRenderer { constructor(canvasId) { this.canvas = document.getElementById(canvasId); this.ctx = this.canvas.getContext('2d'); this.cellWidth = 120; this.cellHeight = 30; } async renderExcelData(excelFile) { const workbook = new ExcelJS.Workbook(); await workbook.xlsx.load(excelFile); const worksheet = workbook.getWorksheet(1); this.renderWorksheet(worksheet); } renderWorksheet(worksheet) { this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height); worksheet.eachRow((row, rowIndex) => { row.eachCell((cell, colIndex) => { this.renderCell(cell, rowIndex - 1, colIndex - 1); }); }); } renderCell(cell, row, col) { const x = col * this.cellWidth; const y = row * this.cellHeight; // 배경색 그리기 if (cell.fill && cell.fill.fgColor) { this.ctx.fillStyle = '#' + cell.fill.fgColor.argb.substr(2); this.ctx.fillRect(x, y, this.cellWidth, this.cellHeight); } // 테두리 그리기 this.ctx.strokeStyle = '#ccc'; this.ctx.strokeRect(x, y, this.cellWidth, this.cellHeight); // 텍스트 스타일 적용 this.ctx.fillStyle = cell.font && cell.font.color ? '#' + cell.font.color.argb.substr(2) : '#000'; this.ctx.font = `${cell.font && cell.font.bold ? 'bold' : 'normal'} 12px Arial`; // 텍스트 그리기 const text = cell.value ? cell.value.toString() : ''; this.ctx.fillText(text, x + 5, y + 20); } // 인터랙티브 기능 추가 addInteractivity() { this.canvas.addEventListener('click', (e) => { const rect = this.canvas.getBoundingClientRect(); const x = e.clientX - rect.left; const y = e.clientY - rect.top; const col = Math.floor(x / this.cellWidth); const row = Math.floor(y / this.cellHeight); console.log(`Cell clicked: ${this.getCellAddress(row, col)}`); }); } getCellAddress(row, col) { return String.fromCharCode(65 + col) + (row + 1); } }

Solution 2: SheetJS + 커스텀 스타일링

// SheetJS + 커스텀 스타일링 엔진 class SheetJSCustomRenderer { constructor(containerId) { this.container = document.getElementById(containerId); this.styleSheet = document.createElement('style'); document.head.appendChild(this.styleSheet); } async loadAndRender(file) { const data = await file.arrayBuffer(); const workbook = XLSX.read(data, { type: 'array' }); // 첫 번째 워크시트 처리 const worksheetName = workbook.SheetNames[0]; const worksheet = workbook.Sheets[worksheetName]; // JSON 형태로 변환 (셀 정보 포함) const jsonData = XLSX.utils.sheet_to_json(worksheet, { header: 1, raw: false, defval: '' }); // 셀 스타일 정보 추출 (제한적) const cellStyles = this.extractCellStyles(worksheet); // HTML 테이블로 렌더링 this.renderAsTable(jsonData, cellStyles); } extractCellStyles(worksheet) { // SheetJS 무료 버전에서는 스타일 정보가 제한적 // 여기서는 셀 내용을 기반으로 스타일을 추정 const styles = {}; const range = XLSX.utils.decode_range(worksheet['!ref']); for (let row = range.s.r; row <= range.e.r; row++) { for (let col = range.s.c; col <= range.e.c; col++) { const cellAddress = XLSX.utils.encode_cell({ r: row, c: col }); const cell = worksheet[cellAddress]; if (cell) { styles[cellAddress] = this.inferStyleFromContent(cell.v); } } } return styles; } inferStyleFromContent(value) { // 내용 기반 스타일 추론 const style = {}; if (typeof value === 'number') { if (value > 1000000) { style.backgroundColor = '#e8f5e8'; // 큰 수치는 연한 녹색 style.fontWeight = 'bold'; } } if (typeof value === 'string') { if (value.includes('총') || value.includes('합계')) { style.backgroundColor = '#fff2cc'; // 총계는 노란색 style.fontWeight = 'bold'; } if (value.length > 20) { style.fontSize = '11px'; // 긴 텍스트는 작은 글씨 } } return style; } renderAsTable(data, styles) { let html = ''; data.forEach((row, rowIndex) => { html += ''; row.forEach((cell, colIndex) => { const cellAddress = XLSX.utils.encode_cell({ r: rowIndex, c: colIndex }); const style = styles[cellAddress] || {}; let cellStyle = 'border: 1px solid #ddd; padding: 8px; '; Object.keys(style).forEach(key => { cellStyle += `${this.camelToKebab(key)}: ${style[key]}; `; }); html += ``; }); html += ''; }); html += '
${cell || ''}
'; this.container.innerHTML = html; } camelToKebab(str) { return str.replace(/([a-z0-9]|(?=[A-Z]))([A-Z])/g, '$1-$2').toLowerCase(); } // MSM 견적서 특화 스타일링 applyMSMQuotationStyles() { const css = ` .msm-quotation table { font-family: 'Malgun Gothic', sans-serif; } .msm-quotation td:nth-child(1) { background-color: #f8f9fa; font-weight: bold; } .msm-quotation td:nth-child(n+3):nth-child(-n+4) { text-align: right; font-family: 'Consolas', monospace; } .msm-quotation tr:last-child td { background-color: #fff2cc; font-weight: bold; border-top: 2px solid #333; } `; this.styleSheet.textContent += css; this.container.className += ' msm-quotation'; } }

Solution 3: 서버-클라이언트 하이브리드 (Node.js)

// 서버사이드 Excel 처리 서비스 (Node.js) const ExcelJS = require('exceljs'); const WebSocket = require('ws'); class ServerExcelProcessor { constructor() { this.wss = new WebSocket.Server({ port: 8080 }); this.setupWebSocketHandlers(); } setupWebSocketHandlers() { this.wss.on('connection', (ws) => { ws.on('message', async (message) => { const data = JSON.parse(message); switch (data.type) { case 'PROCESS_EXCEL': await this.processExcelFile(ws, data.fileData); break; case 'APPLY_FORMATTING': await this.applyFormatting(ws, data.cellData, data.formatting); break; } }); }); } async processExcelFile(ws, fileData) { try { const workbook = new ExcelJS.Workbook(); await workbook.xlsx.load(Buffer.from(fileData, 'base64')); const worksheet = workbook.getWorksheet(1); const processedData = []; worksheet.eachRow((row, rowIndex) => { const rowData = []; row.eachCell((cell, colIndex) => { rowData.push({ value: cell.value, style: this.extractCellStyle(cell), address: cell.address }); }); processedData.push(rowData); }); // 클라이언트로 처리된 데이터 전송 ws.send(JSON.stringify({ type: 'EXCEL_PROCESSED', data: processedData, timestamp: new Date().toISOString() })); } catch (error) { ws.send(JSON.stringify({ type: 'ERROR', message: error.message })); } } extractCellStyle(cell) { return { font: cell.font, fill: cell.fill, border: cell.border, alignment: cell.alignment, numFmt: cell.numFmt }; } async applyFormatting(ws, cellData, formatting) { // 서버에서 고급 서식 처리 후 클라이언트에 결과 전송 const workbook = new ExcelJS.Workbook(); const worksheet = workbook.addWorksheet('Formatted'); // 서식 적용 로직 cellData.forEach(row => { row.forEach(cell => { const excelCell = worksheet.getCell(cell.address); excelCell.value = cell.value; Object.assign(excelCell, formatting); }); }); // 처리된 결과를 클라이언트로 전송 const buffer = await workbook.xlsx.writeBuffer(); ws.send(JSON.stringify({ type: 'FORMATTING_APPLIED', data: buffer.toString('base64') })); } } // 클라이언트사이드 WebSocket 연결 class ClientExcelRenderer { constructor() { this.ws = new WebSocket('ws://localhost:8080'); this.setupWebSocketHandlers(); } setupWebSocketHandlers() { this.ws.onmessage = (event) => { const data = JSON.parse(event.data); switch (data.type) { case 'EXCEL_PROCESSED': this.renderProcessedData(data.data); break; case 'FORMATTING_APPLIED': this.downloadFormattedFile(data.data); break; case 'ERROR': console.error('Server error:', data.message); break; } }; } processFileOnServer(file) { const reader = new FileReader(); reader.onload = (e) => { const base64Data = btoa(String.fromCharCode(...new Uint8Array(e.target.result))); this.ws.send(JSON.stringify({ type: 'PROCESS_EXCEL', fileData: base64Data })); }; reader.readAsArrayBuffer(file); } renderProcessedData(data) { // 서버에서 처리된 데이터를 클라이언트에서 렌더링 const container = document.getElementById('excel-container'); let html = ''; data.forEach(row => { html += ''; row.forEach(cell => { const style = this.styleToCSS(cell.style); html += ``; }); html += ''; }); html += '
${cell.value || ''}
'; container.innerHTML = html; } styleToCSS(styleObj) { let css = 'border: 1px solid #ddd; padding: 6px; '; if (styleObj.fill && styleObj.fill.fgColor) { css += `background-color: #${styleObj.fill.fgColor.argb.substr(2)}; `; } if (styleObj.font) { if (styleObj.font.bold) css += 'font-weight: bold; '; if (styleObj.font.italic) css += 'font-style: italic; '; if (styleObj.font.size) css += `font-size: ${styleObj.font.size}px; `; if (styleObj.font.color) css += `color: #${styleObj.font.color.argb.substr(2)}; `; } return css; } }
